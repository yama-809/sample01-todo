# TODO アプリ Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** バニラHTML/CSS/JavaScriptとlocalStorageを使った、タグフィルター付き個人用TODOウェブアプリを単一HTMLファイルで実装する。

**Architecture:** `index.html` 1ファイルにHTML・CSS・JavaScriptをすべて記述。データ層（純粋関数）とレンダー層（DOM操作）を明確に分離し、テスト可能な構造にする。テストは `test.html` でブラウザ上の簡易アサーションを使って行う。

**Tech Stack:** HTML5, CSS3, Vanilla JavaScript (ES6+), localStorage API

- [ ] **Step 2: データ層の関数を index.html の `<script>` 内に実装する**

`index.html` の `<script>/* Task 3〜5 で追加 */</script>` を以下に置き換える:

```javascript
// --- データ層 ---
const STORAGE_KEY = 'todo-app-tasks';
let tasks = [];
let currentFilter = '';

function loadTasks() {
  try {
    const stored = localStorage.getItem(STORAGE_KEY);
    tasks = stored ? JSON.parse(stored) : [];
  } catch (e) {
    tasks = [];
  }
}

function saveTasks() {
  try {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(tasks));
  } catch (e) { /* プライベートブラウジング等で失敗する場合は無視 */ }
}

function addTask(text, tagsInput) {
  const trimmed = text.trim();
  if (!trimmed) return false;
  const tags = tagsInput
    .split(',')
    .map(t => t.trim())
    .filter(t => t.length > 0);
  tasks.push({
    id: String(Date.now()),
    text: trimmed,
    completed: false,
    tags: tags,
    createdAt: new Date().toISOString()
  });
  saveTasks();
  return true;
}

function deleteTask(id) {
  tasks = tasks.filter(t => t.id !== id);
  saveTasks();
}

function toggleTask(id) {
  const task = tasks.find(t => t.id === id);
  if (task) {
    task.completed = !task.completed;
    saveTasks();
  }
}

// --- レンダー層・イベント層は Task 4〜5 で追加 ---
```

- [ ] **Step 1: レンダー層の関数を追加する**

`index.html` の `// --- レンダー層・イベント層は Task 4〜5 で追加 ---` の行を以下に置き換える:

```javascript
// --- レンダー層 ---
function escapeHtml(str) {
  const div = document.createElement('div');
  div.textContent = str;
  return div.innerHTML;
}

function renderTasks() {
  const list = document.getElementById('task-list');
  const filtered = currentFilter
    ? tasks.filter(t => t.tags.includes(currentFilter))
    : tasks;

  if (filtered.length === 0) {
    list.innerHTML = '<li style="color:#9e9e9e;padding:12px;">タスクがありません</li>';
    return;
  }

  list.innerHTML = filtered.map(task => `
    <li class="task-item${task.completed ? ' completed' : ''}">
      <input type="checkbox" ${task.completed ? 'checked' : ''} data-id="${escapeHtml(task.id)}">
      <span class="task-text">${escapeHtml(task.text)}</span>
      <span class="tags">
        ${task.tags.map(tag => `<span class="tag">${escapeHtml(tag)}</span>`).join('')}
      </span>
      <button class="delete-btn" data-id="${escapeHtml(task.id)}">削除</button>
    </li>
  `).join('');
}

function getAllTags() {
  const tagSet = new Set();
  tasks.forEach(t => t.tags.forEach(tag => tagSet.add(tag)));
  return [...tagSet];
}

function renderTagFilter() {
  const area = document.getElementById('filter-area');
  const tags = getAllTags();
  const buttons = [
    `<button class="filter-btn${currentFilter === '' ? ' active' : ''}" data-tag="">すべて</button>`,
    ...tags.map(tag =>
      `<button class="filter-btn${currentFilter === tag ? ' active' : ''}" data-tag="${escapeHtml(tag)}">${escapeHtml(tag)}</button>`
    )
  ];
  area.innerHTML = buttons.join('');
}

function updateStats() {
  const total = tasks.length;
  const completed = tasks.filter(t => t.completed).length;
  const remaining = total - completed;
  document.getElementById('stats').textContent =
    `残り ${remaining}件 / 完了 ${completed}件`;
}

function render() {
  renderTasks();
  renderTagFilter();
  updateStats();
}

// --- イベント層は Task 5 で追加 ---
```

- [ ] **Step 1: イベントハンドラと init 関数を追加する**

`index.html` の `// --- イベント層は Task 5 で追加 ---` の行を以下に置き換える:

```javascript
// --- イベント層 ---
function init() {
  loadTasks();
  render();

  // タスク追加
  document.getElementById('add-btn').addEventListener('click', () => {
    const textInput = document.getElementById('task-input');
    const tagInput = document.getElementById('tag-input');
    if (addTask(textInput.value, tagInput.value)) {
      textInput.value = '';
      tagInput.value = '';
      render();
      textInput.focus();
    }
  });

  // Enter キーで追加
  document.getElementById('task-input').addEventListener('keydown', e => {
    if (e.key === 'Enter') document.getElementById('add-btn').click();
  });

  // チェックボックスと削除ボタン（イベント委譲）
  document.getElementById('task-list').addEventListener('click', e => {
    if (e.target.matches('input[type="checkbox"]')) {
      toggleTask(e.target.dataset.id);
      render();
    }
    if (e.target.matches('.delete-btn')) {
      deleteTask(e.target.dataset.id);
      render();
    }
  });

  // タグフィルター（イベント委譲）
  document.getElementById('filter-area').addEventListener('click', e => {
    if (e.target.matches('.filter-btn')) {
      currentFilter = e.target.dataset.tag;
      render();
    }
  });
}

document.addEventListener('DOMContentLoaded', init);
```
